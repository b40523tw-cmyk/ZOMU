async function boot() {

async function api(action, payload = {}) {
  if (DEMO) { await sleep(450); return mockApi(action, payload); }

  if (!j.ok && j.code === 'TOKEN_EXPIRED') { liff.logout(); liff.login(); throw new Error('登入過期，重新登入中…'); }
  return j;

function avatar(name, pic, st) {
  const inner = pic ? `<img src="${esc(pic)}" alt="">` : esc(String(name || '?').slice(0, 1));
  return `<div class="av" style="${pic ? '' : 'background:' + colorOf(name)}">${inner}${st ? `<span class="st">${st}</span>` : ''}</div>`;
}
function colorOf(s) { let h = 0; for (const c of String(s)) h = (h * 31 + c.charCodeAt(0)) % 360; return `hsl(${h},45%,62%)`; }
function petGlyph(emoji, img, hue, px) { return art(emoji, img, px, { hue: isUrl(img) ? 0 : hue }); } // 你畫的圖保持原色，變色只用在暫代符號
/* 圖片：一張圖橫排好幾格（例如 3 格）會自動切開。
   mode 'anim' = 循環播放成動圖；'still' = 固定顯示某一格（frame 可用 'last'） */
function art(emoji, img, px, o = {}) {
  const f = o.hue ? `filter:hue-rotate(${o.hue}deg);` : '';
  if (!isUrl(img)) return `<span style="${f}">${esc(emoji)}</span>`;
  return `<span class="spr" data-src="${esc(img)}" data-mode="${o.mode || 'anim'}" data-frame="${o.frame ?? 0}" style="width:${px}px;height:${px}px;${f}"></span>`;
}
const SHEETS = {};
function sheetInfo(src) {
  if (!SHEETS[src]) SHEETS[src] = new Promise(res => {
    const im = new Image();
    im.onload = () => res({ frames: Math.max(1, Math.round(im.naturalWidth / im.naturalHeight)) });
    im.onerror = () => res({ frames: 1, bad: true });
    im.src = src;
  });
  return SHEETS[src];
}
function setFrame(el, i) {
  const n = el._frames || 1;
  el.style.backgroundPosition = n > 1 ? `${(i / (n - 1)) * 100}% 0` : 'center';
  el._cur = i;
}
function hydrate(root = document) {
  root.querySelectorAll('.spr:not([data-ok])').forEach(async el => {
    el.setAttribute('data-ok', '1');
    const src = el.dataset.src, info = await sheetInfo(src);
    el._frames = info.frames;
    el.style.backgroundImage = `url("${src.replace(/"/g, '%22')}")`;
    el.style.backgroundSize = info.frames > 1 ? `${info.frames * 100}% 100%` : 'contain';
    const fr = el.dataset.frame === 'last' ? info.frames - 1 : Math.min(info.frames - 1, +el.dataset.frame || 0);
    setFrame(el, fr);
  });
}
let TICK = 0;
setInterval(() => {
  TICK++;
  document.querySelectorAll('.spr[data-mode="anim"]').forEach(el => {
    const n = el._frames || 1; if (n < 2 || n > 3) return;   // 4 格以上是過程圖（例如蛋），不循環
    const seq = n === 2 ? [0, 0, 0, 1] : n === 3 ? [0, 0, 0, 0, 0, 1, 0, 2] : [0, 0, 0, ...Array.from({ length: n - 1 }, (_, i) => i + 1)];
    setFrame(el, seq[TICK % seq.length]);
  });
}, 170);
function nowMinTaipei() { const d = new Date(); return ((d.getUTCHours() + 8) % 24) * 60 + d.getUTCMinutes(); }
function hm(s) { const m = String(s).match(/(\d+):(\d+)/); return m ? +m[1] * 60 + +m[2] : 510; }
function itemByCode(c) { return (S.items || []).find(i => i.code === c); }

function moodOf() {
  const me = S.me;
  if (me.away) return 'away';
  if (me.leaveToday) return 'capsule';
  if (S.pet && S.pet.stage === '蛋') return me.fedToday ? 'egghappy' : 'egg';
  if (me.fedToday) return 'happy';
  if (me.miss >= 2) return 'sad';
  if (me.miss === 1) return 'hungry';
  return 'waiting';
}

function petSize(p) {
  if (!p) return 100;
  const r = Math.min(1, p.progress / Math.max(1, p.need));
  const R = { '蛋': [92, 92], '幼年': [80, 100], '成長': [100, 120], '成熟': [120, 140], '完全體': [140, 160], '畢業': [150, 150] }[p.stage] || [100, 100];
  return Math.round(R[0] + (R[1] - R[0]) * r);
}
function crackSVG(n) {
  const P = ['M48 26 l4 7 l-5 5 l5 7', 'M28 48 l7 3 l3 -6 l7 5', 'M62 44 l6 -4 l4 7 l7 -3', 'M36 64 l6 -5 l5 6 l6 -6 l6 5', 'M50 52 l-3 6 l5 4 l-2 6'];
  return `<svg class="cracks" viewBox="0 0 100 100">${P.slice(0, n).map(d => `<path d="${d}" fill="none" stroke="#5A4632" stroke-width="2.6" stroke-linecap="round" stroke-linejoin="round"/>`).join('')}</svg>`;
}

function getPos() {
  return new Promise((res, rej) => {
    if (DEMO) return res({ lat: 25.04, lng: 121.56, acc: 18 });
    if (!navigator.geolocation) return rej(new Error('這支手機無法定位'));
    navigator.geolocation.getCurrentPosition(
      p => res({ lat: p.coords.latitude, lng: p.coords.longitude, acc: p.coords.accuracy }),
      e => rej(new Error(e.code === 1 ? '請允許 LINE 使用位置資訊（手機設定 → LINE → 位置）' : '抓不到定位，請到窗邊或稍後再試')),
      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 });
  });
}
async function claim() {
  if (BUSY) return; BUSY = true;
  const btn = $('#feedBtn'); btn.disabled = true; btn.textContent = '📍 正在確認位置…';
  try {
    const pos = await getPos();
    btn.textContent = '🍖 領取中…';
    const j = await api('claim', pos);
    if (!j.ok) { toast(j.error); BUSY = false; render(); return; }
    await feedAnim(j.events);
    NEWER = null; PENDING = j.state;
    await playEvents(j.events);   // 先播動畫（孵化前不先曝光物種），再換新畫面
    S = NEWER || j.state; NEWER = null; PENDING = null; BUSY = false; render();
  } catch (e) { toast(e.message || String(e)); BUSY = false; render(); }
}

      else if (e.up) { confetti(40); await modal(`<div class="big popin">${art(e.emoji, e.img, 130)}</div><h2>🌳 卓木樹長大了！</h2><p>今天全員準時，卓木樹長成了「${esc(e.name)}」！</p><button class="btn full" data-close="1">去看看</button>`); TAB = 'tree'; }
      else toast('🌱 今天全員準時！卓木樹又長大了一點');
    } else if (e.type === 'leave') {
      toast('🛸 寵物進入太空膠囊，今天好好休息！');
    }
  }
}

async function showTableStory(img, emoji) {
  const caps = [
    ['卓木樹結滿了果實', '大家一天一天準時，終於把它養到這麼大。'],

  }
  const info = await sheetInfo(img);
  const steps = Math.max(1, Math.min(3, info.frames));
  const done = modal(`<div class="story"><span class="spr" id="storyArt" data-src="${esc(img)}" data-mode="still" data-frame="0" style="width:250px;height:250px"></span></div>

    <button class="btn full" id="storyNext"${steps === 1 ? ' style="display:none"' : ''}>下一步</button>
    <button class="btn full" data-close="1" id="storyDone"${steps === 1 ? '' : ' style="display:none"'}>坐下來</button>`);
  let i = 0;
  if (steps === 1) confetti(60);
  $('#storyNext').onclick = async () => {
    i++;
    const el = $('#storyArt'); el.style.opacity = 0; await sleep(300);
    setFrame(el, i); el.style.opacity = 1;
    const c = caps[3 - steps + i];

  if (v === 'baby') st.pet = demoPet('owl', '幼年', 6, '咕咕');
  if (v === 'adult') st.pet = demoPet('owl', '成熟', 19, '咕咕');
  if (v === 'final') { st.pet = demoPet('owl', '完全體', 8, '咕咕博士'); me.equipAcc = 'acc_bow'; me.equipBg = 'bg_sunset'; }
  if (v === 'hungry') { me.miss = 1; me.streak = 0; }
  if (v === 'sad') { me.miss = 2; me.streak = 0; }
  if (v === 'away') { me.away = true; me.miss = 3; me.streak = 0; me.homeProgress = 1; }
  if (v === 'capsule') me.leaveToday = true;
  if (v === 'pending') { me.status = '待核准'; st.admin = null; }
  if (v === 'grad') st.pet = demoPet('owl', '完全體', 19, '咕咕博士');
  if (v === 'tree') { st.tree.stage = 5; st.tree.progress = 4; team[3].fed = true; }
  team[0].fed = me.fedToday; team[0].leave = me.leaveToday; team[0].away = me.away; team[0].petEmoji = st.pet.emoji;
  demoBoard(st);
  return st;
}
function demoBoard(st) {
  const vals = { '阿堯': [st.me.streak, st.me.total], '小美': [18, 41], '大熊': [9, 35], '阿哲': [2, 22], '小魚': [6, 30] };
  const rows = st.team.map(m => ({ name: m.name, picture: '', petEmoji: m.petEmoji, petImg: '', petHue: 0, away: m.away, s: vals[m.name][0], t: vals[m.name][1] }));
  st.board = {
    streak: rows.slice().sort((a, b) => b.s - a.s).map(r => Object.assign({}, r, { value: r.s })),
    total: rows.slice().sort((a, b) => b.t - a.t).map(r => Object.assign({}, r, { value: r.t }))
  };
}

function mockApi(action, p) {
  const st = JSON.parse(JSON.stringify(PENDING || S)), me = st.me, ev = [];
  const fail = msg => ({ ok: false, error: msg });
  if (action === 'claim') {
    if (me.fedToday) return fail('今天已經領過飼料了');
    me.fedToday = true; me.streak++; me.total++; me.best = Math.max(me.best, me.streak); me.miss = 0;
    if (me.away) {
      me.homeProgress++;
      if (me.homeProgress >= st.rules.homeNeed) { me.away = false; me.homeProgress = 0; ev.push({ type: 'home', pet: st.pet }); }
      else ev.push({ type: 'awayFeed', progress: me.homeProgress, need: st.rules.homeNeed });
    } else {
      const pet = st.pet; ev.push({ type: 'feed' }); pet.progress++; pet.total++;
      if (pet.progress >= pet.need) {
        if (pet.stage === '蛋') {
          const s = DEMO_SPECIES.find(x => x[0] === 'owl');   // 示範模式固定孵出貓頭鷹
          Object.assign(pet, { species: s[0], speciesName: s[1], emoji: s[2], rarity: s[3], stage: '幼年', progress: 0, need: 20, hue: [0, 0, 30, 160, 280][Math.floor(Math.random() * 5)], intro: '剛剛孵化的新島民', img: (DEMO_IMG[s[0]] || {})['幼年'] || '' });
          ev.push({ type: 'hatch', pet: JSON.parse(JSON.stringify(pet)) });
          const d = st.dex.find(x => x.code === s[0]); if (d) { d.found = true; if (!d.owners.includes('阿堯')) d.owners.push('阿堯'); }
        } else if (pet.stage === '完全體') {
          const g = JSON.parse(JSON.stringify(pet)); g.stage = '畢業'; st.graduates.push(g);
          st.pet = demoPet('', '蛋', 0); ev.push({ type: 'graduate', pet: g });
        } else {
          const from = pet.stage; pet.stage = STAGES[STAGES.indexOf(from) + 1]; pet.progress = 0; pet.img = (DEMO_IMG[pet.species] || {})[pet.stage] || '';
          ev.push({ type: 'evolve', from, to: pet.stage, pet: JSON.parse(JSON.stringify(pet)) });
        }
      }
    }
    if (me.streak % st.rules.itemEvery === 0) {
      const it = st.items.find(i => i.type === '配件' && !me.items.includes(i.code));
      if (it) { me.items.push(it.code); me.equipAcc = it.code; ev.push({ type: 'item', item: it, streak: me.streak }); }
    }
    st.team[0].fed = true; st.team[0].petEmoji = st.pet.emoji;
    if (st.team.every(m => m.fed || m.leave)) {
      st.tree.progress++; st.tree.total++; st.tree.grewToday = true;
      let up = false; if (st.tree.progress >= st.tree.need && st.tree.stage < st.tree.stages.length - 1) { st.tree.stage++; st.tree.progress = 0; up = true; }
      const s = st.tree.stages[st.tree.stage]; ev.push({ type: 'tree', up, stage: st.tree.stage, name: s.name, emoji: s.emoji, img: s.img });
    }
    demoBoard(st);
  } else if (action === 'leave') { me.leaveToday = true; st.team[0].leave = true; ev.push({ type: 'leave' }); }
  else if (action === 'rename') st.pet.name = p.name;
  else if (action === 'equip') {
    if (p.clear) { if (p.type === '背景') me.equipBg = ''; else me.equipAcc = ''; }
    else { const it = st.items.find(i => i.code === p.code); if (it.type === '背景') me.equipBg = it.code; else me.equipAcc = it.code; }
  } else if (action === 'admin_addPlace') {
    if (p.only) st.admin.places.forEach(x => x.on = false);
    const ex = st.admin.places.find(x => x.name === p.name);
    const np = { name: p.name || '現場', lat: p.lat, lng: p.lng, radius: p.radius || 200, on: true };
    if (ex) Object.assign(ex, np); else st.admin.places.push(np);
  } else if (action === 'admin_togglePlace') { const x = st.admin.places.find(x => x.name === p.name); if (x) x.on = p.on; }
  else if (action === 'admin_setWorkday') { st.admin.calendar = st.admin.calendar.filter(c => c.date !== p.date); st.admin.calendar.push({ date: p.date, work: p.work, note: '' }); if (p.date === st.today.date) st.today.workday = p.work; }
  else if (action === 'admin_approve') st.admin.pending = st.admin.pending.filter(x => x.userId !== p.userId);
  return { ok: true, events: ev, state: st };
}
